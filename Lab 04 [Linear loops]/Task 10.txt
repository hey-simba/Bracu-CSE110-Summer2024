import java.util.Scanner;
class question_10
{
    public static void main(String[] args) 
    {
        Scanner sc= new Scanner(System.in);
        System.out.println("Please enter an integer value: ");
        int num= sc.nextInt();
        int store= num;
        int count=0;
        while(num>0)
        {
            num=num/10;
            count+=1;
        }
        int pow= 0;
        int remainder=0;
        while(store>0)
        {
          pow= (int) Math.pow(10,count-1);
          remainder= store/pow;
          System.out.print(remainder);
          store= store%pow;
          count-=1;
          if (count>0)
          {
            System.out.print(",");
          }
            
        }
    }
}
