import java.util.Scanner;
class question_9 
{
    public static void main(String[] args) 
    {
        Scanner sc= new Scanner(System.in);
        System.out.println("Plase enter an integer value: ");
        int num= sc.nextInt();
        int count=0;
        while(num>0)
        {
            num=num/10;
            count+=1;
        }
        System.out.println("Total digits = "+count);
    }
}
