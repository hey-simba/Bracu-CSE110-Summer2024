import java.util.Scanner;
public class question11
{
  public static void main(String[]args)
  {
    Scanner sc= new Scanner(System.in);
    System.out.println("Please enter your first side");
    int side1= sc.nextInt();
    System.out.println("Please enter your second side");
    int side2= sc.nextInt();
    System.out.println("Please enter your third side");
    int side3= sc.nextInt();
    if (side1==side2 && side2==side3)
    {
        System.out.println("This is a Equilateral triangle");
    }
    else if (side1==side2 && side1!=side3)
    {
       System.out.println("This is a Isosceles triangle"); 
    }
    else if (side1==side3 && side1!=side2)
    {
      System.out.println("This is a Isosceles triangle");  
    }
    else if (side2==side3 && side2!=side1)
    {
      System.out.println("This is a Isosceles triangle");  
    }
    else 
    {
        System.out.println("This is a Scalene triangle"); 
    }
  }
}
