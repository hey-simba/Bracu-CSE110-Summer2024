import java.util.Scanner;
public class square
{
    public static void main(String[] agrs)
    {
        Scanner sc= new Scanner(System.in);
        System.out.println("Enter row ");
        int row= sc.nextInt();
        System.out.println("Enter coloumn");
        int column= sc.nextInt();
        for (int i=1; i<=row; i++)
        {
            for (int j=1; j<=column; j++)
            {
                System.out.print(j + " ");
            }
        System.out.println();
        }
    }
}
