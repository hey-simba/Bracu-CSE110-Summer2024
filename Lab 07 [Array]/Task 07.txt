public class Task_7
{
    public static void main(String[]args)
    {
        int arr [] = {23,100,23,56,100};
        
        System.out.println("Input array: ");
        for(int i=0; i<=arr.length-1; i++)
        {
            System.out.print(arr[i]+ " ");
        }
        System.out.println();
        int[] unique= new int[arr.length];
        int unique_num=0;
        
        for (int i=0; i<=arr.length-1; i++)
        {
            int value= arr[i];
            boolean flag= true;
            for (int j=0; j<=unique.length-1; j++)
            {
               int check= unique[j];
               
               if (value==check)
               {
                   flag= false;
                   break;
               }
            }
            if (flag)
            {
                unique[unique_num]= value;
                unique_num+=1;
            }
        }
        for (int i=0 ; i<unique_num ; i++)
        {
            System.out.print(unique[i]+ " ");
        }
    }
}
