
import java.util.Scanner;
public class problem_7
{
    public static void main(String[]agrs)
    {
        Scanner sc= new Scanner(System.in);
        System.out.println("Enter the value of n");
        int n= sc.nextInt();
        //int count= 0;
        int y=0;
        int i= 1;
        while (i<=n)
        {
            if (i%2==0)
            {
              y-= i*2+1; 
            }
            else
            {
              y+= i*2+1;  
            }
            i++;
        }
        System.out.println(y);
        
        
    }
}
