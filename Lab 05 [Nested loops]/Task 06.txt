import java.util.Scanner;
public class Task_6
{
    public static void main(String[] agrs)
    {
        Scanner sc= new Scanner(System.in);
        System.out.println("Enter row: ");
        int row= sc.nextInt();
        
        for (int i=1; i<=row; i++)
        {
            for (int j= 1; j<= row - i; j++)
            {
                System.out.print("  ");
            } 
            for (int k=1; k<=i; k++)
            {
               System.out.print(k + " "); 
            }
        System.out.println();
        }
    }
}
