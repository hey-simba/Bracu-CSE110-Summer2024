import java.util.Scanner;
public class Task_3
{
    public static void main (String[]args)
    {
        Scanner sc= new Scanner(System.in);
        while (true)
        {
            System.out.println("Enter Number: ");
            int num= sc.nextInt();
            int divisor=0;
            if (num%2==0)
            {
                for (int i= 1; i<=num; i++)
                {
                    if (num%i==0)
                    {
                        divisor+=1;
                    }
                    
                }
                System.out.println(num + " has "+ divisor+ " divisors");
                
            }
            divisor= 0;
            if (num%2 !=0)
            {
                break;
            }
        }

    }
}
